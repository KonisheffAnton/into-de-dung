extends SceneTree

var viewport: SubViewport
var demo: HeroHubBattleView
var captures: String
var images: Dictionary = {}
var snapshots: Dictionary = {}
var units: Array[BattleDemoIsoVisual] = []
var all_units: Array[BattleDemoIsoVisual] = []

func _initialize() -> void:
	for argument: String in OS.get_cmdline_user_args():
		if argument.begins_with("--captures="): captures = argument.trim_prefix("--captures=")
	create_timer(40.0).timeout.connect(func() -> void: push_error("Runtime probe timeout"); quit(1))
	call_deferred("_run")

func _run() -> void:
	viewport = SubViewport.new()
	viewport.size = Vector2i(1440, 900)
	viewport.render_target_update_mode = SubViewport.UPDATE_ALWAYS
	root.add_child(viewport)
	var profile := HeroHubState.create(9404, 4)
	profile.enable_mage_deck()
	profile = ExpeditionTransitions.apply(profile, "start").profile
	demo = (load("res://scenes/demo/battle/hero_hub_battle.tscn") as PackedScene).instantiate() as HeroHubBattleView
	var route := HeroHubScreen.new()
	assert(route._configure_first_room_art(demo, profile))
	route.free()
	demo.supplied_session = HeroHubBattleFactory.create(profile).session
	viewport.add_child(demo)
	demo.set_anchors_and_offsets_preset(Control.PRESET_FULL_RECT)
	demo.set_process(false)
	demo.end_turn()
	demo._confirm_selected_mulligan()
	var session := demo.session as BattleCoreReactionsSession
	if session.phase == BattleTacticalSession.Phase.ENEMY_TURN:
		session._enemy_index = session._enemy_ids.size()
		session.step_enemy()
	demo._refresh()
	await _settle()
	var live_before := {"state":ItemSimulationSnapshot.capture(session.state, []).snapshot.to_dict(), "rng":session.context.get_rng().to_dict()}
	var frame := ActionPresentationState.capture(session.state)
	var party: Array[int] = []
	var focus := Vector2.ZERO
	var board := demo.board_view
	for unit: UnitState in session.units:
		if unit.faction != Faction.Type.PLAYER or not unit.is_alive(): continue
		party.append(unit.id)
		focus += board._world_cell_center(unit.cell)
		var row: Dictionary = frame.units[unit.id]
		row.hp = 128.0 if party.size() == 1 else (18.5 if party.size() == 2 else 24.0)
		row.hp_max = 160.0 if party.size() == 1 else 32.0
		row.armor = {"current":64.0 if party.size() == 1 else (7.5 if party.size() == 2 else 16.0),"maximum":80.0 if party.size() == 1 else 24.0,"extra":0.0}
		row.balance = 4.0
		row.balance_max = 12.0
		row.bleeding = {}
		row.elemental = {}
		row.control = {}
		row.shield = {}
		row.raised_shield = {}
	assert(party.size() == 3)
	focus /= 3.0
	board.show_presentation_state(frame)
	board.session = null
	demo._close_character_card()
	var statuses: Array[Dictionary] = [
		{"id":"bleeding","stacks":2,"remaining":3,"embodied":false},
		{"id":"poison","stacks":3,"remaining":2,"embodied":false},
		{"id":"acid","stacks":0,"remaining":2,"embodied":false},
		{"id":"burning","stacks":1,"remaining":3,"embodied":false},
		{"id":"wet","stacks":0,"remaining":2,"embodied":false}]
	for index: int in party.size():
		var visual := board._presentation_visual(party[index])
		visual.status_entries = statuses.slice(0, 1 + index * 2)
		visual.set_process(false)
		visual.hovered = false
		units.append(visual)
	for visual: BattleDemoIsoVisual in board._entity_visuals:
		visual.set_process(false)
		for node: Node in visual.find_children("*", "AnimatedSprite2D", true, false): (node as AnimatedSprite2D).stop()
	for visual: BattleDemoIsoVisual in board._entity_visuals:
		if visual.unit != null and visual.unit.alive: all_units.append(visual)
	assert(all_units.size() == 5)
	units[2].hovered = true
	var samples: Dictionary = {}
	var matches := 0
	var modes: Array[Dictionary] = [{"name":"100", "zoom":1.0}, {"name":"75", "zoom":0.75}, {"name":"canvas-113", "zoom":1.0}]
	for mode: Dictionary in modes:
		var screen_focus := Vector2(720, 340) - board.global_position
		board.zoom_camera(float(mode.zoom), screen_focus)
		board.pan_camera(screen_focus - board.world_to_view(focus))
		await _settle()
		if mode.name == "canvas-113":
			# Только canvas меняется: board polling должен инвалидировать экранные draw-команды.
			viewport.canvas_transform = Transform2D(Vector2(1.13, 0), Vector2(0, 1.13), Vector2(13.37, 7.61))
			await _settle()
		await RenderingServer.frame_post_draw
		var img := viewport.get_texture().get_image()
		DirAccess.make_dir_recursive_absolute(captures)
		assert(img.save_png(captures.path_join("battle-" + String(mode.name) + ".png")) == OK)
		images[mode.name] = img
		var points: Array[Vector2] = []
		for index: int in all_units.size():
			var visual := all_units[index]
			var origin := visual.get_visual_offset() + visual.get_unit_bars_offset() * visual.grid_config.tile_size.x / 128.0
			var point := (visual.get_viewport_transform() * visual.get_global_transform() * origin).round()
			points.append(point)
			for track: int in 3:
				var rect := Rect2i(Vector2i(point) + Vector2i(-24, [1,12,23][track]), Vector2i(48, 9 if track < 2 else 4))
				var pixels := img.get_region(rect).get_data()
				var key := "%d:%d" % [index,track]
				if mode.name == "100":
					samples[key] = pixels
				else:
					assert(pixels == samples[key], "Screen-sized track drift: " + String(mode.name) + ":" + key)
					matches += 1
		snapshots[mode.name] = points
	assert(matches == 30)
	assert(await _dense_neighbor_probe(), "Status row occludes a visible neighbor")
	assert(live_before.state == ItemSimulationSnapshot.capture(session.state, []).snapshot.to_dict() and live_before.rng == session.context.get_rng().to_dict())
	var report := FileAccess.open(captures.path_join("probe.json"), FileAccess.WRITE)
	report.store_string(JSON.stringify({"status":"PASS", "core_rng_unchanged":true, "camera_and_canvas_rgba_tracks":matches, "production_renderer":true, "neighbor_body_masks_checked":8, "neighbor_row_occlusions":0, "positions":snapshots, "fixture":"five native classes; synthetic presentation resources; horizontal hover row3+N immediately above armor; five native classes"}, "\t"))
	report.close()
	viewport.queue_free()
	await process_frame
	print("STATUS READABILITY RUNTIME: PASS;30 fixed vitals camera/canvas RGBA matches;8 neighbor RGBA masks unchanged;core/RNG unchanged")
	quit(0)

func _settle() -> void:
	for frame: int in 6: await process_frame

func _dense_neighbor_probe() -> bool:
	viewport.canvas_transform = Transform2D.IDENTITY
	var board := demo.board_view
	var target := units[2]
	var center := target.position
	var neighbors: Array[BattleDemoIsoVisual] = []
	var offsets: Array[Vector2] = [Vector2(64,48),Vector2(-64,48),Vector2(64,-48),Vector2(-64,-48)]
	for visual: BattleDemoIsoVisual in all_units:
		visual.hovered = false
		visual.half_steps = 0
		# sync_scene_art повторно включает _process у скинов; стоп idle нужен до синхронизации.
		(visual as BattleManagedUnitVisual).ambient_animation_paused = true
		visual.sync_scene_art()
		visual.set_process(false)
		if visual == target: continue
		visual.position = center + offsets[neighbors.size()]
		neighbors.append(visual)
	for zoom: float in [1.0, 0.75]:
		var focus := Vector2(720,400) - board.global_position
		board.zoom_camera(zoom, focus)
		board.pan_camera(focus - board.world_to_view(center))
		target.hovered = true
		# Одинаковые уголки hover в обоих кадрах; сравнивается только новый ряд.
		target._status_hud.hide()
		await _settle()
		await RenderingServer.frame_post_draw
		var baseline := viewport.get_texture().get_image()
		baseline.save_png(captures.path_join("dense-baseline-" + str(zoom) + ".png"))
		var masks: Array[Array] = []
		for neighbor: BattleDemoIsoVisual in neighbors:
			var coverage := await _body_coverage(neighbor)
			coverage.save_png(captures.path_join("body-alpha-" + str(neighbor.unit.id) + "-" + str(zoom) + ".png"))
			if neighbor.visual_class_id == &"rogue" and zoom == 1.0:
				print("Rogue alpha at previous differing pixel744,376: ", coverage.get_pixel(744,376).a)
			var foot := neighbor.get_viewport_transform() * neighbor.get_global_transform() * neighbor.get_visual_offset()
			var bounds := Rect2i(Vector2i(foot + Vector2(-55,-100)*zoom),Vector2i(Vector2(110,140)*zoom)).intersection(Rect2i(Vector2i.ZERO,baseline.get_size()))
			var pixels: Array[Vector2i] = []
			for y: int in range(bounds.position.y,bounds.end.y):
				for x: int in range(bounds.position.x,bounds.end.x):
					var pixel := Vector2i(x,y)
					# Полупрозрачный AA-край смешивается с подложкой ПОЗАДИ спрайта по общим правилам.
					if coverage.get_pixelv(pixel).a == 1.0: pixels.append(pixel)
			if pixels.size() <= 20:
				printerr("FAIL: blank neighbor mask for ", neighbor.visual_class_id)
				quit(1)
				return false
			masks.append(pixels)
		target._status_hud.show()
		await _settle()
		await RenderingServer.frame_post_draw
		var hovered := viewport.get_texture().get_image()
		assert(hovered.save_png(captures.path_join("dense-" + ("100" if zoom == 1.0 else "75") + ".png")) == OK)
		var row_bounds := target.status_hud_bounds()
		var transform := target._status_hud.get_global_transform_with_canvas()
		var row_screen := Rect2(transform * row_bounds.position, row_bounds.size * zoom).grow(1.0)
		for mask_index: int in masks.size():
			var mask: Array = masks[mask_index]
			for pixel: Vector2i in mask:
				# Прежние уголки hover вне ряда не относятся к новому рисунку статусов.
				if row_screen.has_point(Vector2(pixel)) and hovered.get_pixelv(pixel) != baseline.get_pixelv(pixel):
					printerr("Status row paints over a visible neighbor: " + JSON.stringify({"pixel":pixel,"target_position":target.position,"target_offset":target.get_visual_offset(),"row":row_screen,"neighbor":neighbors[mask_index].visual_class_id,"neighbor_position":neighbors[mask_index].position,"baseline":baseline.get_pixelv(pixel),"hovered":hovered.get_pixelv(pixel)}))
					quit(1)
					return false
	return true

func _body_coverage(visual: BattleDemoIsoVisual) -> Image:
	var mask_viewport := SubViewport.new()
	mask_viewport.size = viewport.size
	mask_viewport.transparent_bg = true
	mask_viewport.render_target_update_mode = SubViewport.UPDATE_ALWAYS
	root.add_child(mask_viewport)
	var body := visual.get_node("ArtRoot") as Node2D
	var sources: Array[Node] = body.find_children("*", "Sprite2D", true, false)
	sources.append_array(body.find_children("*", "AnimatedSprite2D", true, false))
	for node: Node in sources:
		var original := node as Node2D
		if not original.is_visible_in_tree(): continue
		var clone := Sprite2D.new()
		if original is Sprite2D:
			var sprite := original as Sprite2D
			clone.texture = sprite.texture
			clone.centered = sprite.centered
			clone.offset = sprite.offset
			clone.flip_h = sprite.flip_h
			clone.flip_v = sprite.flip_v
			clone.hframes = sprite.hframes
			clone.vframes = sprite.vframes
			clone.frame = sprite.frame
			clone.region_enabled = sprite.region_enabled
			clone.region_rect = sprite.region_rect
		else:
			var sprite := original as AnimatedSprite2D
			clone.texture = sprite.sprite_frames.get_frame_texture(sprite.animation,sprite.frame)
			clone.centered = sprite.centered
			clone.offset = sprite.offset
			clone.flip_h = sprite.flip_h
			clone.flip_v = sprite.flip_v
		clone.material = original.material
		clone.texture_filter = original.texture_filter
		clone.modulate = original.modulate * original.self_modulate * body.modulate
		mask_viewport.add_child(clone)
		clone.transform = original.get_viewport_transform() * original.get_global_transform()
	await _settle()
	await RenderingServer.frame_post_draw
	var result := mask_viewport.get_texture().get_image()
	mask_viewport.queue_free()
	await process_frame
	return result
