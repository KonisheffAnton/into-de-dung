extends SceneTree

var viewport: SubViewport
var demo: HeroHubBattleView
var captures: String
var images: Dictionary = {}
var snapshots: Dictionary = {}
var units: Array[BattleDemoIsoVisual] = []
var all_units: Array[BattleDemoIsoVisual] = []
var smooth_font: FontFile
var smooth_size: int = 10
var smooth_offset: float = 0.0
var ink_height: int = 0
var pixel_matches: int = 0

func _initialize() -> void:
	for argument: String in OS.get_cmdline_user_args():
		if argument.begins_with("--captures="): captures = argument.trim_prefix("--captures=")
	call_deferred("_run")

func _run() -> void:
	viewport = SubViewport.new()
	viewport.size = Vector2i(1440, 900)
	viewport.render_target_update_mode = SubViewport.UPDATE_ALWAYS
	root.add_child(viewport)
	var profile := HeroHubState.create(9404, 4)
	profile.enable_mage_deck()
	profile = ExpeditionTransitions.apply(profile, "start").profile
	demo = (load("res://scenes/demo/battle/hero_hub_battle.tscn") as PackedScene).instantiate() as HeroHubBattleView
	var route := HeroHubScreen.new()
	assert(route._configure_first_room_art(demo, profile))
	route.free()
	demo.supplied_session = HeroHubBattleFactory.create(profile).session
	viewport.add_child(demo)
	demo.set_anchors_and_offsets_preset(Control.PRESET_FULL_RECT)
	demo.set_process(false)
	demo.end_turn()
	demo._confirm_selected_mulligan()
	var session := demo.session as BattleCoreReactionsSession
	if session.phase == BattleTacticalSession.Phase.ENEMY_TURN:
		session._enemy_index = session._enemy_ids.size()
		session.step_enemy()
	demo._refresh()
	await _settle()
	var live_before := {"state":ItemSimulationSnapshot.capture(session.state, []).snapshot.to_dict(), "rng":session.context.get_rng().to_dict()}
	var frame := ActionPresentationState.capture(session.state)
	var party: Array[int] = []
	var focus := Vector2.ZERO
	var board := demo.board_view
	for unit: UnitState in session.units:
		if unit.faction != Faction.Type.PLAYER or not unit.is_alive(): continue
		party.append(unit.id)
		focus += board._world_cell_center(unit.cell)
		var row: Dictionary = frame.units[unit.id]
		row.hp = 128.0 if party.size() == 1 else (18.5 if party.size() == 2 else 24.0)
		row.hp_max = 160.0 if party.size() == 1 else 32.0
		row.armor = {"current":64.0 if party.size() == 1 else (7.5 if party.size() == 2 else 16.0),"maximum":80.0 if party.size() == 1 else 24.0,"extra":0.0}
		row.balance = 4.0
		row.balance_max = 12.0
		row.bleeding = {}
		row.elemental = {}
		row.control = {}
		row.shield = {}
		row.raised_shield = {}
	assert(party.size() == 3)
	focus /= 3.0
	board.show_presentation_state(frame)
	board.session = null
	demo._close_character_card()
	var statuses: Array[Dictionary] = [
		{"id":"bleeding","stacks":2,"remaining":3,"embodied":false},
		{"id":"poison","stacks":3,"remaining":2,"embodied":false},
		{"id":"acid","stacks":0,"remaining":2,"embodied":false},
		{"id":"burning","stacks":1,"remaining":3,"embodied":false},
		{"id":"wet","stacks":0,"remaining":2,"embodied":false}]
	for index: int in party.size():
		var visual := board._presentation_visual(party[index])
		visual.status_entries = statuses.slice(0, 1 + index * 2)
		visual.set_process(false)
		visual.hovered = false
		units.append(visual)
	for visual: BattleDemoIsoVisual in board._entity_visuals:
		visual.set_process(false)
		for node: Node in visual.find_children("*", "AnimatedSprite2D", true, false): (node as AnimatedSprite2D).stop()
	await _prepare_smooth_font()
	for visual: BattleDemoIsoVisual in board._entity_visuals:
		if visual.unit == null or not visual.unit.alive: continue
		all_units.append(visual)
		visual.set_meta(&"mock_fixed_vitals", true)
		visual.set_meta(&"mock_compact_c", true)
		visual.hovered = false
		var pen := Node2D.new()
		pen.name = "FixedVitalsMock"
		pen.z_as_relative = false
		pen.z_index = RenderLayers.z_index_for(RenderLayers.Layer.UI_PREVIEW) + 1
		visual.add_child(pen)
		pen.draw.connect(_draw_fixed_vitals.bind(visual, pen))
		visual._status_hud.z_as_relative = false
		visual._status_hud.z_index = pen.z_index
	assert(all_units.size() == 5)
	var samples: Dictionary = {}
	for zoom: float in [1.0, 0.75]:
		var screen_focus := Vector2(720, 340) - board.global_position
		board.zoom_camera(zoom, screen_focus)
		board.pan_camera(screen_focus - board.world_to_view(focus))
		for visual: BattleDemoIsoVisual in all_units:
			visual.queue_redraw()
			visual._status_hud.queue_redraw()
			(visual.get_node("FixedVitalsMock") as Node2D).queue_redraw()
		await _settle()
		await RenderingServer.frame_post_draw
		var label := str(roundi(zoom * 100))
		var img := viewport.get_texture().get_image()
		DirAccess.make_dir_recursive_absolute(captures)
		assert(img.save_png(captures.path_join("battle-" + label + ".png")) == OK)
		images[label] = img
		var positions: Array[Vector2] = []
		for visual: BattleDemoIsoVisual in units: positions.append(_bar_anchor(visual))
		snapshots[label] = positions
		for index: int in all_units.size():
			var point := Vector2i(_bar_anchor(all_units[index]))
			for track: int in 3:
				var rect := Rect2i(point + Vector2i(-24, [1, 12, 23][track]), Vector2i(48, 9 if track < 2 else 4))
				var crop := img.get_region(rect)
				var key := "%d:%d" % [index, track]
				if zoom == 1.0: samples[key] = crop.get_data()
				else:
					assert(samples[key] == crop.get_data(), "Screen pixels changed with zoom: " + key)
					pixel_matches += 1
				for x: int in crop.get_width():
					assert(crop.get_pixel(x, 0) == Color.BLACK and crop.get_pixel(x, crop.get_height() - 1) == Color.BLACK)
				for y: int in crop.get_height():
					assert(crop.get_pixel(0, y) == Color.BLACK and crop.get_pixel(crop.get_width() - 1, y) == Color.BLACK)
	assert(pixel_matches == 15)
	assert(live_before.state == ItemSimulationSnapshot.capture(session.state, []).snapshot.to_dict() and live_before.rng == session.context.get_rng().to_dict())
	await _comparison()
	var report := FileAccess.open(captures.path_join("probe.json"), FileAccess.WRITE)
	report.store_string(JSON.stringify({"status":"PASS", "fixture":"catalog4 seed9404;5 native classes; synthetic ally DTO128/64,18.5/7.5,24/16", "core_rng_unchanged":true, "screen_bar_size":[48,9], "screen_font_size":smooth_size, "glyph_ink_height":ink_height, "font_baseline_offset":smooth_offset, "font":"Combat Lettering / Play Bold700, GRAY AA, NORMAL hinting", "tracks_pixel_identical_100_75":pixel_matches, "all_track_borders_black":true, "positions":snapshots, "scope":"P only; fixed-screen vitals and compact C statuses not integrated; overlay above terrain"}, "\t"))
	report.close()
	viewport.queue_free()
	await process_frame
	print("SMOOTH FIXED VITALS PROBE: PASS;3 PNG;15 track RGBA matches100/75; core/RNG unchanged")
	quit(0)

func _bar_anchor(visual: BattleDemoIsoVisual) -> Vector2:
	return (visual.get_global_transform_with_canvas() * (visual.get_visual_offset() + visual.get_unit_bars_offset() * visual.grid_config.tile_size.x / 128.0)).round()

func _draw_fixed_vitals(visual: BattleDemoIsoVisual, pen: Node2D) -> void:
	# Только иллюстрация: отдельный экранный рисунок не меняет world/picking/состояние.
	pen.draw_set_transform_matrix(pen.get_global_transform_with_canvas().affine_inverse())
	var point := _bar_anchor(visual)
	var row: BattleUnitView = visual.unit
	var armor: Dictionary = visual.armor_values
	var armor_rect := Rect2(point + Vector2(-24, 1), Vector2(48, 9))
	var hp_rect := Rect2(point + Vector2(-24, 12), Vector2(48, 9))
	_draw_track(pen, armor_rect, float(armor.current) / float(armor.maximum) if float(armor.maximum) > 0 else 0, Color("9ca5a6"), Color("b1b9b6"), Color("828d8f"), Color("747d80"))
	_draw_track(pen, hp_rect, row.hp / row.hp_max if row.hp_max > 0 else 0, Color("829969"), Color("96ab7c"), Color("6e8259"), Color("777d70"))
	_draw_track(pen, Rect2(point + Vector2(-24, 23), Vector2(48, 4)), BattleBalanceDisplay.ratio(row.balance, row.balance_max), Color("bea16c"), Color("cdb888"), Color("a58b60"), Color("45494b"))
	_draw_amount(pen, armor_rect, BattleEventText.amount_text(float(armor.current)))
	_draw_amount(pen, hp_rect, BattleEventText.amount_text(row.hp))
	pen.draw_set_transform_matrix(Transform2D.IDENTITY)

func _draw_track(pen: Node2D, rect: Rect2, ratio: float, fill: Color, light: Color, shade: Color, empty: Color) -> void:
	pen.draw_rect(rect, Color.BLACK)
	var inner := rect.grow(-1)
	pen.draw_rect(inner, empty)
	if ratio <= 0: return
	var filled := Rect2(inner.position, Vector2(minf(inner.size.x, maxf(1, inner.size.x * clampf(ratio, 0, 1))), inner.size.y))
	pen.draw_rect(filled, fill)
	pen.draw_rect(Rect2(filled.position, Vector2(filled.size.x, 1)), light)
	pen.draw_rect(Rect2(filled.end - Vector2(filled.size.x, 1), Vector2(filled.size.x, 1)), shade)

func _draw_amount(pen: Node2D, rect: Rect2, text: String) -> void:
	var width := smooth_font.get_string_size(text, HORIZONTAL_ALIGNMENT_LEFT, -1, smooth_size).x
	assert(width <= 44)
	var baseline := Vector2(rect.get_center().x - width * 0.5, rect.get_center().y + smooth_offset).round()
	pen.draw_string(smooth_font, baseline, text, HORIZONTAL_ALIGNMENT_LEFT, -1, smooth_size, Color.BLACK)

func _prepare_smooth_font() -> void:
	smooth_font = (load("res://assets/ui/combat_fonts/combat_letters.ttf") as FontFile).duplicate() as FontFile
	smooth_font.antialiasing = TextServer.FONT_ANTIALIASING_GRAY
	smooth_font.hinting = TextServer.HINTING_NORMAL
	smooth_font.subpixel_positioning = TextServer.SUBPIXEL_POSITIONING_DISABLED
	smooth_font.oversampling = 1.0
	smooth_font.keep_rounding_remainders = false
	var measure := SubViewport.new()
	measure.size = Vector2i(256, 200)
	measure.transparent_bg = true
	measure.render_target_update_mode = SubViewport.UPDATE_ALWAYS
	root.add_child(measure)
	var pen := Node2D.new()
	measure.add_child(pen)
	pen.draw.connect(func() -> void:
		for size: int in range(8, 12):
			pen.draw_string(smooth_font, Vector2(10, (size - 8) * 50 + 35), "0123456789.", HORIZONTAL_ALIGNMENT_LEFT, -1, size, Color.BLACK)
	)
	pen.queue_redraw()
	await _settle()
	await RenderingServer.frame_post_draw
	var img := measure.get_texture().get_image()
	for size: int in range(8, 12):
		var ink := img.get_region(Rect2i(0, (size - 8) * 50, 256, 50)).get_used_rect()
		if ink.size.y <= 7:
			smooth_size = size
			ink_height = ink.size.y
			smooth_offset = 35.0 - ink.position.y - ink.size.y * 0.5
	assert(ink_height >= 6 and ink_height <= 7)
	measure.queue_free()
	await process_frame

func _settle() -> void:
	for index: int in 6: await process_frame

func _comparison() -> void:
	var sheet := SubViewport.new()
	sheet.size = Vector2i(1600, 1040)
	sheet.render_target_update_mode = SubViewport.UPDATE_ALWAYS
	root.add_child(sheet)
	var bg := ColorRect.new()
	bg.color = Color("171b21")
	bg.size = Vector2(sheet.size)
	sheet.add_child(bg)
	_label(sheet, "НЕПИКСЕЛЬНЫЕ ЦИФРЫ · ПОЛОСЫ 9 ЭКРАННЫХ ПИКСЕЛЕЙ", Vector2(28, 16), 25)
	_label(sheet, "Поле отдаляется · размер полос и цифр сохраняется · Play Bold со сглаживанием", Vector2(28, 54), 19)
	for column: int in 2:
		var label := "100" if column == 0 else "75"
		var left := 24.0 + column * 790.0
		_label(sheet, "Масштаб поля " + label + "%", Vector2(left, 105), 24)
		_label(sheet, "Фрагмент ×2 · полосы 48×9 px · цифры %d px по высоте" % ink_height, Vector2(left, 140), 17)
		var full: Image = images[label]
		_picture(sheet, full.get_region(Rect2i(572, 205, 320, 250)), Vector2(left + 46, 177), 2)
		_label(sheet, "Детали цифр ×4 · одинаковые значения", Vector2(left, 701), 20)
		var points: Array = snapshots[label]
		for index: int in 3:
			var p: Vector2 = points[index]
			_picture(sheet, full.get_region(Rect2i(Vector2i(p) + Vector2i(-24, 1), Vector2i(48, 26))), Vector2(left + index * 240, 738), 4)
		_label(sheet, "Исходный размер захвата · без увеличения", Vector2(left, 869), 18)
		for index: int in 3:
			var p: Vector2 = points[index]
			_picture(sheet, full.get_region(Rect2i(Vector2i(p) + Vector2i(-24, 1), Vector2i(48, 26))), Vector2(left + 76 + index * 240, 904), 1)
	_label(sheet, "Макет, не внедрён. Цвета прежние; числа тестовые. При 75% полосы крупнее относительно персонажей.", Vector2(28, 981), 18)
	await _settle()
	await RenderingServer.frame_post_draw
	assert(sheet.get_texture().get_image().save_png(captures.path_join("comparison.png")) == OK)
	sheet.queue_free()
	await process_frame

func _picture(parent: Node, img: Image, at: Vector2, factor: float) -> void:
	var picture := TextureRect.new()
	picture.texture = ImageTexture.create_from_image(img)
	picture.texture_filter = CanvasItem.TEXTURE_FILTER_NEAREST
	picture.position = at
	picture.scale = Vector2.ONE * factor
	parent.add_child(picture)

func _label(parent: Node, text: String, at: Vector2, font_size: int) -> void:
	var label := Label.new()
	label.text = text
	label.position = at
	label.add_theme_font_size_override("font_size", font_size)
	parent.add_child(label)
